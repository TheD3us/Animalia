import {
  Component,
  Input,
  Output,
  EventEmitter,
  ChangeDetectionStrategy,
  HostBinding,
  OnInit,
  ElementRef,
  Renderer2
} from '@angular/core';

@Component({
  selector: 'app-star-rating',
  templateUrl: './star-rating.component.html',
  styleUrls: ['./star-rating.component.scss'],
  changeDetection: ChangeDetectionStrategy.OnPush
})
export class StarRatingComponent implements OnInit {
  /** Valeur actuelle (1..maxStars) */
  @Input() value = 0;

  /** Nombre d'étoiles (par défaut 5) */
  @Input() maxStars = 5;

  /** Si true, le composant est lecture seule */
  @Input() readonly = false;

  /** Taille CSS (ex: '24px') */
  @Input() size = '28px';

  /** Evénement quand la note change (nombre entier) */
  @Output() valueChange = new EventEmitter<number>();

  /** étoile survolée actuellement (prévisualisation) */
  hoverValue = 0;

  /** index focus (1..maxStars) pour clavier */
  focusedIndex = 0;

  stars: number[] = [];

  constructor(private el: ElementRef, private renderer: Renderer2) {}

  ngOnInit(): void {
    this.maxStars = Math.max(1, Math.floor(this.maxStars || 5));
    this.stars = Array.from({ length: this.maxStars }, (_, i) => i + 1);
    this.value = this.clampValue(this.value);
    // expose tabindex if not readonly for accessibility
    if (!this.readonly) {
      this.renderer.setAttribute(this.el.nativeElement, 'tabindex', '0');
      this.renderer.setAttribute(this.el.nativeElement, 'role', 'radiogroup');
    } else {
      this.renderer.setAttribute(this.el.nativeElement, 'aria-hidden', 'false');
    }
  }

  clampValue(v: number) {
    if (!Number.isFinite(v)) return 0;
    return Math.max(0, Math.min(this.maxStars, Math.floor(v)));
  }

  // Hover events (desktop)
  onMouseEnter(index: number) {
    if (this.readonly) return;
    this.hoverValue = index;
  }
  onMouseLeave() {
    if (this.readonly) return;
    this.hoverValue = 0;
  }

  // Click / touch to set
  setValue(index: number) {
    if (this.readonly) return;
    const newVal = this.clampValue(index);
    if (newVal !== this.value) {
      this.value = newVal;
      this.valueChange.emit(this.value);
    }
  }

  // Keyboard handling on the root element
  onKeydown(event: KeyboardEvent) {
    if (this.readonly) return;
    const key = event.key;
    const left = key === 'ArrowLeft' || key === 'ArrowDown';
    const right = key === 'ArrowRight' || key === 'ArrowUp';
    if (left || right) {
      event.preventDefault();
      const delta = left ? -1 : 1;
      const next = this.clampValue((this.hoverValue || this.value) + delta);
      this.hoverValue = 0;
      this.setValue(next);
    } else if (key >= '1' && key <= String(this.maxStars)) {
      event.preventDefault();
      this.setValue(Number(key));
    } else if (key === ' ' || key === 'Enter') {
      event.preventDefault();
      // if nothing selected, set to 1
      const idx = this.focusedIndex || (this.value || 1);
      this.setValue(idx);
    }
  }

  // focus handling per-star for aria/radio-like behaviour
  onStarFocus(index: number) {
    if (this.readonly) return;
    this.focusedIndex = index;
  }
  onStarBlur() {
    if (this.readonly) return;
    this.focusedIndex = 0;
  }

  // Used by template to know if star should be filled
  isFilled(index: number) {
    const active = this.hoverValue || this.value;
    return index <= active;
  }
}
